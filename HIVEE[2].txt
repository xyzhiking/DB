📜 Step 3: Open Hive CLI or Hue Query Editor

Option A — CLI:

hive


CREATE DATABASE IF NOT EXISTS flight_db;
USE flight_db;

-- Drop if already exists
DROP TABLE IF EXISTS flight_info;
DROP TABLE IF EXISTS airline_info;
DROP TABLE IF EXISTS flight_data_ext;


CREATE TABLE flight_info (
  flight_id STRING,
  airline STRING,
  origin STRING,
  destination STRING,
  departure_date DATE,
  departure_delay INT,
  arrival_delay INT
)
ROW FORMAT DELIMITED
FIELDS TERMINATED BY ','
STORED AS TEXTFILE;

-- Alter table to add new column

ALTER TABLE flight_info ADD COLUMNS (aircraft_type STRING);


CREATE EXTERNAL TABLE IF NOT EXISTS flight_data_ext (
  flight_id STRING,
  airline STRING,
  origin STRING,
  destination STRING,
  departure_date DATE,
  departure_delay INT,
  arrival_delay INT

)
ROW FORMAT DELIMITED
FIELDS TERMINATED BY ','
STORED AS TEXTFILE
LOCATION '/user/hive/flight_data/';

🔗 Step 8: Join Tables (Example)




SELECT f.flight_id, f.airline, a.country, f.departure_date
FROM flight_info f
JOIN airline_info a
ON f.airline = a.airline;

Note: Hive indexing is outdated, use partitions instead.

AS 'COMPACT'
WITH DEFERRED REBUILD;

ALTER INDEX idx_departure_date ON flight_info REBUILD;



📊 Step 10: Average Departure Delay per Day in 2008

SELECT departure_date, AVG(departure_delay) AS avg_dep_delay
FROM flight_info
WHERE YEAR(departure_date) = 2008
GROUP BY departure_date
ORDER BY departure_date;
